import pandas as pd
import plotly.express as px
import plotly.graph_objects as go

# Configuración de página
st.set_page_config(
    page_title="Dilema del Prisionero - Cátedra Procesos Psicológicos Básicos",
    page_icon="⚖️",
    layout="wide"
)

# Ocultar menú predeterminado de Streamlit para pantalla limpia
st.markdown("""
    <style>
    #MainMenu {visibility: hidden;}
    footer {visibility: hidden;}
    header {visibility: hidden;}
    </style>
""", unsafe_allow_html=True)

# Estructura de almacenamiento en memoria compartida
@st.cache_resource
def get_global_db():
    return []

db = get_global_db()

# Título Principal
st.title("⚖️ Dilema del Prisionero: Experimento en Vivo")
st.caption("Cátedra I de Procesos Psicológicos Básicos | Unidad de Cognición Social & Toma de Decisiones")

# Selector de Rol en la barra lateral
st.sidebar.header("🕹️ Menú de Navegación")
modo = st.sidebar.radio("Selecciona tu rol:", ["📱 Modo Alumno (Responder)", "📊 Modo Docente (Resultados en Vivo)"])

# -----------------------------------------------------------------------------
# MODO ALUMNO
# -----------------------------------------------------------------------------
if modo == "📱 Modo Alumno (Responder)":
    st.subheader("Formulario de Decisión Individual")
    
    # Consigna completa con reglas de condena
    st.info("""
    **Consigna del Ejercicio:**
    
    Tú y tu compañero/a anónimo han sido detenidos por la policía como sospechosos de un delito. Se encuentran incomunicados en celdas separadas. El fiscal les ofrece a cada uno de forma independiente las siguientes dos opciones:
    * **NO CONFESAR** (no delatar a tu compañero/a).
    * **CONFESAR** (delatar a tu compañero/a).

    **Reglas de Condena:**
    * **Si ambos (Jugador A y B) deciden NO CONFESAR:** Ambos jugadores reciben **1 año de prisión** cada uno.
    * **Si un Jugador decide CONFESAR y el otro NO CONFESAR:** El jugador que confiesa recibe **0 años de prisión**. El jugador que no confiesa recibe **20 años de prisión**.
    * **Si ambos (Jugador A y B) deciden CONFESAR:** Ambos jugadores reciben **5 años de prisión** cada uno.
    """)
    
    with st.form("form_alumno", clear_on_submit=True):
        col1, col2 = st.columns(2)
        with col1:
            comision = st.selectbox("Número de Comisión:", [f"Comisión {i}" for i in range(1, 61)])
        with col2:
            id_pareja = st.number_input("Número de ID de Pareja asignado (Ej: 1, 2, 3...):", min_value=1, max_value=200, value=1, step=1)
            
        rol_jugador = st.radio("Rol asignado en la pareja:", ["Jugador 1", "Jugador 2"], horizontal=True)
        
        st.markdown("---")
        st.markdown("### 1. ¿Qué decisión tomás?")
        eleccion = st.radio("Selecciona tu acción:", ["CONFESAR", "NO CONFESAR"], key="q1")
        
        st.markdown("### 2. ¿Qué decisión crees que tomará tu compañero/a?")
        prediccion_1st = st.radio("Predicción de 1.er Orden (ToM 1st):", ["CONFESAR", "NO CONFESAR"], key="q2")
        
        st.markdown("### 3. ¿Qué decisión crees que tu compañero/a piensa que vos tomarás?")
        prediccion_2nd = st.radio("Predicción de 2.º Orden (ToM 2nd):", ["CONFESAR", "NO CONFESAR"], key="q3")
        
        submitted = st.form_submit_button("📤 Enviar Decisión", use_container_width=True)
        
        if submitted:
            registro = {
                "comision": comision,
                "id_pareja": int(id_pareja),
                "rol": rol_jugador,
                "eleccion": eleccion,
                "prediccion_1st": prediccion_1st,
                "prediccion_2nd": prediccion_2nd
            }
            # Actualizar si ya envió previamente o agregar nuevo
            existing = [i for i, r in enumerate(db) if r["comision"] == comision and r["id_pareja"] == id_pareja and r["rol"] == rol_jugador]
            if existing:
                db[existing[0]] = registro
            else:
                db.append(registro)
                
            st.success(f"¡Decisión registrada correctamente para la {comision}, Pareja ID {id_pareja} ({rol_jugador})!")
            st.balloons()

# -----------------------------------------------------------------------------
# MODO DOCENTE
# -----------------------------------------------------------------------------
else:
    st.subheader("📊 Panel del Docente - Resultados en Vivo")
    
    col_f1, col_f2 = st.columns(2)
    with col_f1:
        comision_sel = st.selectbox("Filtrar por Comisión:", ["Todas las Comisiones"] + [f"Comisión {i}" for i in range(1, 61)])
    with col_f2:
        if st.button("🗑️ Limpiar Datos (Reset)", help="Borra las respuestas grabadas"):
            db.clear()
            st.rerun()

    df_raw = pd.DataFrame(db)
    
    if df_raw.empty:
        st.info("Aún no se han recibido respuestas. Pide a los alumnos que envíen sus decisiones.")
    else:
        if comision_sel != "Todas las Comisiones":
            df_filtered = df_raw[df_raw["comision"] == comision_sel]
        else:
            df_filtered = df_raw.copy()
            
        st.metric("Total de Respuestas Recibidas", len(df_filtered))
        
        # Agrupar por parejas
        parejas_dict = {}
        for _, row in df_filtered.iterrows():
            key = (row["comision"], row["id_pareja"])
            if key not in parejas_dict:
                parejas_dict[key] = {}
            parejas_dict[key][row["rol"]] = row
            
        cuadrantes = {
            ("NO CONFESAR", "NO CONFESAR"): 0,
            ("CONFESAR", "NO CONFESAR"): 0,
            ("NO CONFESAR", "CONFESAR"): 0,
            ("CONFESAR", "CONFESAR"): 0
        }
        
        parejas_completas = []
        aciertos_tom1 = 0
        aciertos_tom2 = 0
        total_eval_tom = 0
        
        for (com, id_p), jug in parejas_dict.items():
            if "Jugador 1" in jug and "Jugador 2" in jug:
                j1 = jug["Jugador 1"]
                j2 = jug["Jugador 2"]
                
                e1, e2 = j1["eleccion"], j2["eleccion"]
                p1_1st, p2_1st = j1["prediccion_1st"], j2["prediccion_1st"]
                p1_2nd, p2_2nd = j1["prediccion_2nd"], j2["prediccion_2nd"]
                
                cuadrantes[(e1, e2)] = cuadrantes.get((e1, e2), 0) + 1
                
                # Evaluación ToM 1er orden: ¿P1 predijo la elección real de P2?
                if p1_1st == e2: aciertos_tom1 += 1
                if p2_1st == e1: aciertos_tom1 += 1
                
                # Evaluación ToM 2do orden: ¿P1 predijo lo que P2 creía que P1 elegiría?
                if p1_2nd == p2_1st: aciertos_tom2 += 1
                if p2_2nd == p1_1st: aciertos_tom2 += 1
                
                total_eval_tom += 2
                
                # Matriz de condenas (0, 1, 5, 20 años)
                if e1 == "NO CONFESAR" and e2 == "NO CONFESAR":
                    pena1, pena2, res = "1 año", "1 año", "Cooperación Mutua"
                elif e1 == "CONFESAR" and e2 == "NO CONFESAR":
                    pena1, pena2, res = "LIBRE (0 años)", "20 años", "Deserción J1"
                elif e1 == "NO CONFESAR" and e2 == "CONFESAR":
                    pena1, pena2, res = "20 años", "LIBRE (0 años)", "Deserción J2"
                else:
                    pena1, pena2, res = "5 años", "5 años", "Deserción Mutua (Equilibrio Nash)"
                    
                parejas_completas.append({
                    "Comisión": com,
                    "ID Pareja": id_p,
                    "J1 Elección": e1,
                    "J2 Elección": e2,
                    "J1 Condena": pena1,
                    "J2 Condena": pena2,
                    "J1 ToM 1st": "Acierto" if p1_1st == e2 else "Error",
                    "J2 ToM 1st": "Acierto" if p2_1st == e1 else "Error",
                    "J1 ToM 2nd": "Acierto" if p1_2nd == p2_1st else "Error",
                    "J2 ToM 2nd": "Acierto" if p2_2nd == p1_1st else "Error",
                    "Resultado Pareja": res
                })
        
        col_g1, col_g2 = st.columns(2)
        
        with col_g1:
            st.markdown("### 🎯 Matriz de Decisiones 2x2 (Parejas Completas)")
            z_vals = [
                [cuadrantes[("NO CONFESAR", "CONFESAR")], cuadrantes[("CONFESAR", "CONFESAR")]],
                [cuadrantes[("NO CONFESAR", "NO CONFESAR")], cuadrantes[("CONFESAR", "NO CONFESAR")]]
            ]
            fig_matrix = go.Figure(data=go.Heatmap(
                z=z_vals,
                x=["J2: NO CONFESAR", "J2: CONFESAR"],
                y=["J1: NO CONFESAR", "J1: CONFESAR"],
                text=[
                    [f"J2 Traiciona: {z_vals[0][0]}", f"Deserción Mutua (5y): {z_vals[0][1]}"],
                    [f"Cooperación Mutua (1y): {z_vals[1][0]}", f"J1 Traiciona: {z_vals[1][1]}"]
                ],
                texttemplate="%{text}",
                textfont={"size": 14},
                colorscale="Blues",
                showscale=False
            ))
            fig_matrix.update_layout(height=350, margin=dict(l=10, r=10, t=30, b=10))
            st.plotly_chart(fig_matrix, use_container_width=True)
            
        with col_g2:
            st.markdown("### 🧠 Exactitud en Teoría de la Mente (ToM)")
            if total_eval_tom > 0:
                pct1 = (aciertos_tom1 / total_eval_tom) * 100
                pct2 = (aciertos_tom2 / total_eval_tom) * 100
                
                m1, m2 = st.columns(2)
                m1.metric("ToM 1.er Orden (Lectura de Acción)", f"{pct1:.1f}%")
                m2.metric("ToM 2.º Orden (Creencia Recíproca)", f"{pct2:.1f}%")
                
                df_tom_chart = pd.DataFrame({
                    "Nivel ToM": ["1.er Orden (Acción del otro)", "2.º Orden (Pensamiento recíproco)"],
                    "Aciertos (%)": [pct1, pct2]
                })
                fig_tom = px.bar(df_tom_chart, x="Nivel ToM", y="Aciertos (%)", color="Nivel ToM", text_auto=".1f", range_y=[0, 100])
                fig_tom.update_layout(height=280, showlegend=False, margin=dict(l=10, r=10, t=10, b=10))
                st.plotly_chart(fig_tom, use_container_width=True)
            else:
                st.info("Esperando que ambas partes de las parejas envíen sus datos para calcular ToM.")
                
        st.markdown("### 📋 Detalle por Parejas Emparejadas")
        if parejas_completas:
            df_parejas = pd.DataFrame(parejas_completas)
            st.dataframe(df_parejas, use_container_width=True)
            
            csv_data = df_parejas.to_csv(index=False).encode('utf-8')
            st.download_button("📥 Descargar Reporte Completo en CSV", csv_data, "dilema_prisionero_resultados.csv", "text/csv")
        else:
            st.warning("Hay respuestas registradas, pero aún se necesita completar ambas partes (Jugador 1 y Jugador 2) de al menos una pareja.")
